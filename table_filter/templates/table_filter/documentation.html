راهنمای کامل فیلتر اکسلی

سیستم پیشرفته فیلتر، مرتب‌سازی و جستجو برای جداول

معرفی

فیلتر اکسلی یک سیستم پیشرفته و کاملاً سفارشی‌سازی‌شده برای اضافه کردن قابلیت‌های فیلتر، مرتب‌سازی و جستجو به هر جدولی است. این سیستم الهام‌گرفته از فیلترهای Microsoft Excel است و تمام قابلیت‌های مورد نیاز برای مدیریت داده‌های جدولی را فراهم می‌کند.

نکته: این سیستم به صورت کاملاً خودکار تمام ستون‌ها را تشخیص می‌دهد و بر اساس نوع داده، فیلترهای مناسب ایجاد می‌کند.

ویژگی‌ها

فیلتر پیشرفته

انتخاب چندگانه مقادیر با checkbox، جستجوی سریع در مقادیر، و فیلتر بر اساس نوع داده

مرتب‌سازی

مرتب‌سازی صعودی و نزولی برای تمام انواع داده‌ها شامل متن، عدد، تاریخ

جستجوی سریع

جستجوی لحظه‌ای در مقادیر ستون‌ها با پشتیبانی از متن فارسی

کنترل نمایش

امکان مخفی/نمایش کردن ستون‌ها و تنظیم ترتیب آن‌ها

ذخیره وضعیت

حفظ فیلترها، مرتب‌سازی و تنظیمات کاربر در localStorage

واکنش‌گرا

طراحی کاملاً واکنش‌گرا برای موبایل و تبلت

نصب و راه‌اندازی

1 اضافه کردن فایل‌های CSS و JS

فایل‌های مورد نیاز را در template خود include کنید:

{% load static %}

<!-- CSS فیلتر -->
<link rel="stylesheet" href="{% static 'table_filter/css/table_filter.css' %}">

<!-- JavaScript فیلتر -->
<script src="{% static 'table_filter/js/table_filter.js' %}"></script>

2 تنظیم ساختار جدول

جدول خود را با attribute های مناسب تنظیم کنید:

<table class="table" id="myTable">
    <thead>
        <tr>
            <th class="column-id" data-column-type="number" data-column-id="id">شناسه</th>
            <th class="column-name" data-column-type="text" data-column-id="name">نام</th>
            <th class="column-amount" data-column-type="number" data-column-id="amount">مبلغ</th>
            <th class="column-date" data-column-type="date" data-column-id="date">تاریخ</th>
            <th class="column-status" data-column-type="status" data-column-id="status">وضعیت</th>
        </tr>
    </thead>
    <tbody>
        <!-- ردیف‌های جدول -->
    </tbody>
</table>

3 راه‌اندازی JavaScript

در انتهای صفحه، فیلتر را راه‌اندازی کنید:

<script>
document.addEventListener('DOMContentLoaded', function() {
    const tableFilter = new TableFilter({
        tableSelector: '#myTable',
        enableToast: true,
        enableIndicators: true,
        enableSummary: true,
        enableClearAll: true,
        autoDetectColumns: true,
        
        columnConfig: {
            id: { type: 'number', filterable: true, sortable: true },
            name: { type: 'text', filterable: true, sortable: true },
            amount: { type: 'number', filterable: true, sortable: true },
            date: { type: 'date', filterable: true, sortable: true },
            status: { type: 'status', filterable: true, sortable: true }
        }
    });
});
</script>

تنظیمات پیشرفته

انواع ستون‌ها

نوع ستون کاربرد قابلیت‌ها
text متن‌های عادی جستجوی متنی، فیلتر checkbox، مرتب‌سازی الفبایی
number اعداد و مبالغ مرتب‌سازی عددی، فیلتر محدوده‌ای، پشتیبانی از اعداد فارسی
date تاریخ‌ها مرتب‌سازی تاریخی، فیلتر بازه زمانی، پشتیبانی از تاریخ شمسی
status وضعیت‌ها و برچسب‌ها فیلتر چندگانه، گروه‌بندی وضعیت‌ها
boolean مقادیر صحیح/غلط فیلتر True/False، نمایش به صورت ✓/✗
choice گزینه‌های از پیش تعریف شده فیلتر بر اساس لیست محدود گزینه‌ها

تنظیمات کانفیگ

const tableFilter = new TableFilter({
    // انتخابگر جدول
    tableSelector: '#myTable',
    
    // فعال‌سازی toast notification
    enableToast: true,
    
    // نمایش نشانگر فیلترهای فعال
    enableIndicators: true,
    
    // نمایش خلاصه فیلترهای اعمال شده
    enableSummary: true,
    
    // دکمه پاک کردن همه فیلترها
    enableClearAll: true,
    
    // تشخیص خودکار نوع ستون‌ها
    autoDetectColumns: true,
    
    // کلاس دکمه فیلتر
    filterButtonClass: 'filter-dropdown-btn',
    
    // راه‌اندازی خودکار
    autoInit: true,
    
    // تنظیمات سفارشی ستون‌ها
    columnConfig: {
        columnId: {
            type: 'text|number|date|status|boolean|choice',
            filterable: true,
            sortable: true,
            name: 'نام نمایشی ستون'
        }
    }
});

مثال‌های کاربردی

1. جدول ساده

<table class="table" id="simpleTable">
    <thead>
        <tr>
            <th class="column-name">نام</th>
            <th class="column-email">ایمیل</th>
            <th class="column-city">شهر</th>
        </tr>
    </thead>
    <tbody>
        <tr>
            <td>علی احمدی</td>
            <td>ali@example.com</td>
            <td>تهران</td>
        </tr>
    </tbody>
</table>

<script>
// راه‌اندازی ساده - تشخیص خودکار
new TableFilter({
    tableSelector: '#simpleTable'
});
</script>

2. جدول پیشرفته با تنظیمات سفارشی

<script>
const advancedFilter = new TableFilter({
    tableSelector: '#advancedTable',
    enableToast: true,
    enableIndicators: true,
    enableSummary: true,
    enableClearAll: true,
    
    columnConfig: {
        purchase_id: {
            type: 'number',
            name: 'شناسه خرید',
            filterable: true,
            sortable: true
        },
        customer_name: {
            type: 'text',
            name: 'نام مشتری',
            filterable: true,
            sortable: true
        },
        purchase_amount: {
            type: 'number',
            name: 'مبلغ خرید',
            filterable: true,
            sortable: true
        },
        purchase_date: {
            type: 'date',
            name: 'تاریخ خرید',
            filterable: true,
            sortable: true
        },
        status: {
            type: 'status',
            name: 'وضعیت',
            filterable: true,
            sortable: true
        }
    }
});

// متدهای کاربردی
advancedFilter.clearAllFilters(); // پاک کردن همه فیلترها
advancedFilter.getFilteredRowCount(); // دریافت تعداد ردیف‌های فیلتر شده
</script>

سوالات متداول

کافی است فایل‌های CSS و JS را include کنید و در JavaScript کد new TableFilter({tableSelector: '#yourTable'}) را اضافه کنید.

بله، سیستم کاملاً از زبان فارسی، اعداد فارسی، و تاریخ شمسی پشتیبانی می‌کند.

تنظیمات فیلتر، مرتب‌سازی و نمایش ستون‌ها در localStorage مرورگر ذخیره می‌شود و در بازدیدهای بعدی بازیابی می‌شود.
